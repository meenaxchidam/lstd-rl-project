\section{Problem statement and setup}

\begin{frame}{Problem statement and motivation}
  \speaker{\memberA}
  \small
  \begin{itemize}
    \item An agent moves from state to state and receives a reward after each move. A policy is the rule that chooses those moves.
    \item In this project the policy is already chosen and does not change, so the system is a Markov reward process (MRP).
    \item If every transition probability were known, that estimate would be the solution of a linear system.
    \item The probabilities are unknown. We only see sampled trajectories, so the system has to be built from data.
    \item TD($\lambda$) builds it through repeated step-size updates. The step size is chosen by hand, and a poor choice makes learning slow or unstable.
    \item Boyan (ICML 1999, \hl{Best Paper Award}) replaces those updates with one least-squares solve, LSTD($\lambda$). The same trajectories give a more accurate value, and there is no step size.
    \item With only a few trajectories the least-squares matrix can be poorly determined. \hl{Ridge regression}, $(A+\epsilon I)\vth=b$, is our addition for that case.
  \end{itemize}
\end{frame}

\section{Background}

\begin{frame}{Policy evaluation}
  \speaker{\memberA}
  \small
  \setlength{\abovedisplayskip}{0.3em}%
  \setlength{\belowdisplayskip}{0.25em}%
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Exact value method}
    \begin{itemize}
      \item The value of a state is its expected total future reward:
      \[ V^\pi(s)=\mathbb{E}\Big[\sum_{t\ge 0}\gamma^t r_t \;\Big|\; s_0=s\Big] \]
      \item The Bellman equation splits that into one reward plus the rest:
      \[ V(s)=\mathbb{E}\big[r+\gamma V(s')\big] \]
      \item When $P$ and the rewards are known, this is a linear system and can be solved directly.
    \end{itemize}
    \column{0.50\textwidth}
    \textbf{Value-function approximation}
    \begin{itemize}
      \item In this project the transitions are unknown. Each step is only a sample $(s,\, r,\, s_{\mathrm{next}},\, \mathrm{done})$.
      \item One stored number per state does not scale, so $d$ weights represent every state, with $d\ll|S|$:
      \[ \hat V(s)=\vphi(s)^\top\vth,\qquad \vphi(s)\in\mathbb{R}^d \]
      \item The features $\vphi(s)$ are fixed in advance. Monte Carlo, TD, and LSTD are three ways to learn $\vth$.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Monte Carlo, TD, and LSTD}
  \speaker{\memberA}
  \scriptsize
  All three approximate the same value function. Write $\hat V(s)=\vphi(s)^\top\vth$ and $\vphi_t=\vphi(s_t)$.
  \vspace{0.2em}
  \setlength{\abovedisplayskip}{3pt}%
  \setlength{\belowdisplayskip}{3pt}%
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.305\textwidth}
      \begin{minipage}[t][4.35cm][t]{0.94\linewidth}
      \raggedright
      {\color{accent}\small\bfseries Monte Carlo}\par\vspace{0.28em}
      The target is the return actually received. The update waits until the episode ends.
      \begin{align*}
        G_t &= \textstyle\sum_{k\ge 0}\gamma^k r_{t+k+1}\\
        e_t &= G_t-\hat V(s_t)\\
        \vth &\leftarrow \vth+\alpha\,e_t\,\vphi_t
      \end{align*}
      Uses the observed outcome.\par
      One odd episode can move $\hat V$ a long way.\par
      $\lambda=1$ is this end of the scale.
      \end{minipage}
    \end{column}
    \begin{column}{0.028\textwidth}
      \centering\textcolor{hair}{\rule{0.7pt}{4.35cm}}
    \end{column}
    \begin{column}{0.324\textwidth}
      \begin{minipage}[t][4.35cm][t]{0.94\linewidth}
      \raggedright
      {\color{accent}\small\bfseries TD($\lambda$)}\par\vspace{0.28em}
      The target is one reward plus a discounted estimate of the next state.
      \begin{align*}
        \delta_t &= r_t+\gamma\hat V(s_{t+1})\\
                 &\quad {}-\hat V(s_t)\\
        \vz_t &= \gamma\lambda\,\vz_{t-1}+\vphi_t\\
        \vth &\leftarrow \vth+\alpha\,\delta_t\,\vz_t
      \end{align*}
      $\lambda=0$: only the current state.\par
      Larger $\lambda$: earlier states keep a decaying share.\par
      $\alpha$ sets how far each nudge goes.
      \end{minipage}
    \end{column}
    \begin{column}{0.028\textwidth}
      \centering\textcolor{hair}{\rule{0.7pt}{4.35cm}}
    \end{column}
    \begin{column}{0.305\textwidth}
      \begin{minipage}[t][4.35cm][t]{0.94\linewidth}
      \raggedright
      {\color{accent}\small\bfseries LSTD($\lambda$)}\par\vspace{0.28em}
      Same $\delta_t$ and $\vz_t$. Solve for the $\vth$ at which the updates add to zero.
      \begin{align*}
        \Delta_t &= \vphi_t-\gamma\vphi_{t+1}\\
        A &\mathrel{+}= \vz_t\Delta_t^\top\\
        b &\mathrel{+}= r_t\,\vz_t\\
        \vth &= A^{-1}b
      \end{align*}
      No step size.\par
      Terminal step: $\vphi_{t+1}=\mathbf{0}$.\par
      LSTD($1$) = Monte Carlo regression.
      \end{minipage}
    \end{column}
  \end{columns}
\end{frame}

\section{TD($\lambda$)}

\begin{frame}{Why move from Monte Carlo to TD?}
  \speaker{\memberB}
  \small
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{Monte Carlo: wait for the episode}
      \[G_t=r_t+\gamma r_{t+1}+\cdots\]
      Uses the actual return: unbiased on average. But a long random future can make the target noisy, and learning waits until termination.
    \end{block}
    \column{0.48\textwidth}
    \begin{block}{Temporal difference: update each step}
      \[r_t+\gamma\hat V(s_{t+1})\]
      Bootstraps from the next-state estimate. It can update before termination and often has lower target variance, at the cost of bias while that estimate is inaccurate.
    \end{block}
  \end{columns}
  \begin{keyidea}
    \small
    \textbf{Corridor example:} $A\to B$ gives 0; $B$ pays 1 with probability .75. In our sample, A appears once and that episode loses, so MC's one return is 0. TD can use $0+\hat V(B)$; as B is learned, it passes that information back to A. MC is still correct in the long run; TD needs fewer direct A visits here.
  \end{keyidea}
\end{frame}

\begin{frame}{TD($\lambda$): the update and the role of $\lambda$}
  \speaker{\memberB}
  \small
  \[\hat V(s)=\vphi(s)^\top\vth,\qquad
    \delta_t=r_t+\gamma\hat V(s_{t+1})-\hat V(s_t),\qquad
    \vz_t=\gamma\lambda\vz_{t-1}+\vphi(s_t)\]
  \[\vth\leftarrow\vth+\alpha_n\sum_t\delta_t\vz_t\qquad\text{(paper-style trajectory update)}\]
    \begin{itemize}
      \item $\vz_t$ shares each TD error with recently active features.
      \item $\lambda=0$: one-step TD target.
      \item $0<\lambda<1$: mixture of multi-step targets.
      \item $\lambda=1$: Monte Carlo return.
      \item The paper's key TD interpretation is this bias--variance continuum, not three separate ``views.''
    \end{itemize}
  \note{Focus here on TD updates, eligibility traces, and the role of lambda. The LSTD motivation slide will introduce the expected-update fixed point and show why it can be solved directly.}
\end{frame}

\begin{frame}[noframenumbering]{What Boyan's paper says about TD}
  \speaker{\memberB}
  \small
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Convergence.} With linear value functions, TD converges under a suitable diminishing step-size schedule. The paper's sufficient conditions are $\alpha_n\geq0$, $\sum_n\alpha_n=\infty$, and $\sum_n\alpha_n^2<\infty$.

    \vspace{0.5em}
    \textbf{Limitations.} After each trajectory, ordinary TD keeps its weights but no transition statistics. Reusing old data requires replay. Results are sensitive to step-size choice, initial weights, and feature scaling.

    \column{0.48\textwidth}
    \textbf{Paper experiment.} Six decaying step-size schedules; $\lambda\in\{0,0.2,0.4,0.6,0.8,1\}$; 10 trials. The schedule is $\alpha_n=a_0\frac{n_0+1}{n_0+n}$, with $a_0\in\{0.01,0.1\}$ and $n_0\in\{100,1000,10^6\}$.

    \vspace{0.5em}
    \begin{block}{Finding}
      The paper reports that TD performance depends critically on the step-size schedule. Its tested schedules meet the convergence conditions, yet their finite-data learning curves differ.
    \end{block}
  \end{columns}
  \vspace{0.2em}
  \scriptsize This slide summarizes the paper's TD analysis; the preceding experiment slide reports our TD-only 20-seed run at $\lambda=0.4$.
\end{frame}

\section{LSTD($\lambda$)}

\begin{frame}{Motivation for LSTD}
  \speaker{\memberC}
  \begin{itemize}
    \item For a fixed policy, learn weights in $\hat V(s)=\vphi(s)^\top\vth$.
    \item TD approaches a fixed point through incremental updates. Where should those weights settle?
    \item Averaged over trajectories, the update has a linear form:
      \[ \mathbb{E}[\Delta\vth\mid\vth]=\alpha\,(\mathbf{d}+C\vth). \]
      Here $\mathbf{d}$ summarizes rewards and $C\vth$ the value-prediction contribution.
    \item At the fixed point, the \emph{expected} corrections balance:
      \[ \boxed{\mathbf{d}+C\vth=0}. \]
  \end{itemize}
  \begin{keyidea}
    \textbf{LSTD:} estimate this balance from experience and solve it directly.
  \end{keyidea}
  \note{Bridge from the TD section: we have seen the update rule and step-size sensitivity; now ask where those updates settle. TD approaches a fixed point through stochastic updates, and the step-size schedule affects the path and rate of convergence. For the paper's episode-wise TD, weights stay fixed within an episode. Under the standard coverage, feature and step-size assumptions, the expected system has a unique stable solution. Individual sample corrections need not vanish. Source: Boyan (1999), Sections 1--2, Equation 1.}
\end{frame}

\begin{frame}{The LSTD($\lambda$) algorithm (paper Fig.~2)}
  \speaker{\memberC}
  \begin{columns}[T]
    \column{0.54\textwidth}
    \begin{block}{LSTD($\lambda$)}
      \small
      $A \leftarrow 0$, $\mathbf{b} \leftarrow 0$\\
      for each trajectory:\\
      \quad $\vz \leftarrow 0$\\
      \quad for each step $t$:\\
      \qquad $\vz \leftarrow \gamma\lambda\vz + \vphi_t$\\
      \qquad $A \leftarrow A + \vz\,(\vphi_t - \gamma\vphi_{t+1})^\top$\\
      \qquad $\mathbf{b} \leftarrow \mathbf{b} + \vz\, r_t$\\
      when weights are needed: solve $A\vth=\mathbf{b}$
    \end{block}
    \column{0.43\textwidth}
    \begin{itemize}
      \item $\vz$ remembers previously visited features within the episode.
      \item Keep $A,\mathbf{b}$ across episodes. Reset only $\vz$.
      \item After $n$ episodes: $\mathbf{b}/n\approx\mathbf{d}$ and $A/n\approx-C$.
      \item \hl{No TD step size or initial weight guess}
      \item Zero next-state features at termination.
    \end{itemize}
  \end{columns}
  \vspace{0.3em}
  \small Implementation: \texttt{solve}; SVD-based least-squares fallback if the direct solve fails.
  \note{Explain the sign: A uses current minus discounted next features, whereas C uses the opposite order. The sample system estimates the expected TD fixed point. A is K by K; b, z and theta have length K. The paper uses gamma=1; the slide includes discounting. Statistics retain each observation's contribution, not a recoverable copy of every trajectory. Source: Boyan (1999), Equation 2 and Figure 2; src/lstd.py.}
\end{frame}

\begin{frame}{Why use LSTD instead of TD?}
  \speaker{\memberC}
  \begin{columns}[T]
    \column{0.43\textwidth}
    \begin{itemize}
      \item \textbf{Data efficiency:} fewer trajectories than tested TD schedules on the Boyan chain.
      \item \textbf{No TD step-size schedule:} solve the collected system directly.
      \item Useful when observations are costly and the feature count is modest.
    \end{itemize}
    \column{0.54\textwidth}
    \begin{block}{Same corridor, now with LSTD}
      \small
      {\centering
        \begin{tikzpicture}
          \node[st,minimum size=5mm,label=below:{\scriptsize corridor}] (A) {A};
          \node[st,minimum size=5mm,right=1.2cm of A,label=below:{\scriptsize machine}] (B) {B};
          \node[draw=ink,thick,rounded corners,right=1.2cm of B,font=\small] (E) {end};
          \draw[arr] (A) -- node[above]{\scriptsize $r=0$} (B);
          \draw[arr] (B) -- node[above]{\scriptsize $r=0$ or $1$} (E);
        \end{tikzpicture}
      \par}\vspace{0.4em}
      \textbf{Recall the sample:} A was visited once (return 0); B paid 1 on 6 of its 8 visits.\\[0.4em]
      \textbf{LSTD(0), one pass and solve:}\\
      $\hat V(B)=6/8=0.75$,\quad $\hat V(A)=0+\hat V(B)=0.75$.\\[0.4em]
      TD(0) propagates this gradually; LSTD(0) solves both values together.
    \end{block}
  \end{columns}
  \note{Continue the corridor already introduced in the TD section, rather than introducing a new environment. Recall that MC gives A the one observed return of 0, while TD can propagate B's estimate back to A. Now ask how LSTD obtains that same empirical TD fixed point directly. The diagram is a visual reminder: A gives no reward and leads to B; B pays 0 or 1 and terminates; gamma=1. The first episode starts at A and has payout 0. The other seven start at B, with six wins and one zero, so there are eight visits to B in total. LSTD(0) estimates B as 6/8 and solves A=0+B, obtaining .75 for both. Ordinary TD starts with B unknown when it first visits A; replay lets the improved estimate of B reach A through later updates. This is an illustrative fixed dataset evaluated with our implementations, not Boyan's experimental environment. Monte Carlo and LSTD(1) fit the one observed return from A as 0. About 2000 deterministic replays with alpha=.01 yield approximately .7499, not an exact or universal replay requirement. The paper's data-efficiency result is on the Boyan chain; it does not guarantee universal superiority or lower runtime. Sources: Boyan (1999), Sections 2, 4--5; docs/MC_vs_TD_vs_LSTD.md.}
\end{frame}

\begin{frame}{Recursive LSTD}
  \speaker{\memberC}
  \begin{itemize}
    \item Maintain $B=A^{-1}$ to avoid repeated full solves.
    \item Each transition changes $A$ by one outer product:
      \[ A_{\mathrm{new}}=A+\vz\mathbf{u}^{\top},\qquad
         \mathbf{u}=\vphi_t-\gamma\vphi_{t+1}. \]
  \end{itemize}
  \begin{block}{Sherman--Morrison inverse update}
    \[\boxed{B\leftarrow B-\frac{B\vz\,\mathbf{u}^{\top}B}{1+\mathbf{u}^{\top}B\vz}}\]
    Then $\mathbf{b}\leftarrow\mathbf{b}+\vz r_t$ and obtain weights using $\vth=B\mathbf{b}$.
  \end{block}
  \begin{keyidea}
    Maintain the inverse instead of solving from scratch.
  \end{keyidea}
  \note{An inverse of the initially zero A does not exist. Our implementation initializes B0=delta I, equivalent to A0=I/delta and batch regularization epsilon=1/delta. Here delta is an initialization parameter, not the TD error. Recursive and batch results match when their regularization is aligned, subject to numerical precision and valid inverse updates. The paper mentions recursive least-squares as an option; our code implements Sherman--Morrison explicitly. Sources: Boyan (1999), end of Section 2; src/lstd.py, recursive_lstd.}
\end{frame}

\begin{frame}{Conclusions and limitations}
  \speaker{\memberD}
  \footnotesize
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Conclusions}
    \begin{itemize}
      \item \hl{Reproduced Boyan (1999):} LSTD($\lambda$) beats every TD setting we tried, at every $n$ and $\lambda$ ($2$--$3.5\times$ below the best TD at $n=100$ and $10^4$).
      \item TD's best step size shifts with the data budget; a poor choice costs $20$--$40\times$ more error.
      \item LSTD has no step size: it solves $A\vth=\mathbf{b}$ directly, reusing every transition.
      \item $\lambda$ barely matters for LSTD here: the features are exact.
      \item Price: $O(K^2)$ per step vs.\ $O(K)$ ($K=256$: 64\,ms vs.\ 2.7\,ms).
      \item \hl{Ridge LSTD} halves the error with one trajectory when features are weak; $\le 3\%$ gain when exact.
    \end{itemize}
    \column{0.42\textwidth}
    \textbf{Limitations}
    \begin{itemize}
      \item Small chains (13 and 19 states); Boyan's features are exact, the best case for LSTD.
      \item Prediction only (fixed policy); control, e.g.\ LSPI, not tested.
      \item TD tuned over a 12-setting grid, with the paper's per-trajectory updates.
      \item $O(K^2)$ memory and time limit LSTD for very large feature sets.
      \item Ridge: $\epsilon$ picked after the fact; fixed $\epsilon$ weakens as $n$ grows; shrinks toward $\vth=0$.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Backup: likely questions}
  \footnotesize
  \begin{description}[Why more data-efficient?]
    \item[Why more data-efficient?] It keeps every transition in $A$, $\mathbf{b}$ and solves exactly. TD uses each sample for one small step and then forgets it. LSTD is effectively model-based.
    \item[What if $A$ is singular?] Too little data or redundant features. Paper: SVD pseudo-inverse. Us: \texttt{solve} with SVD fallback, or ridge $(A+\epsilon I)$.
    \item[Cost in $K$?] $O(K^2)$ per step, $O(K^3)$ per solve (or $O(K^2)$ per step with recursive LSTD), vs.\ $O(K)$ for TD. Many features + cheap data $\Rightarrow$ TD can win on wall-clock time.
    \item[LSTD with $\lambda=1$?] Linear regression of features onto Monte-Carlo returns.
    \item[Control?] Yes: LSTD-Q inside policy iteration = LSPI (Lagoudakis \& Parr, 2003).
    \item[Is the comparison fair?] Same episodes per seed; TD gets its best of 12 configs at every $n$.
    \item[Why does ridge not help on Boyan?] $A$ is well conditioned, and shrinkage biases $\vth$ toward 0 while $\vth^* = (-24,-16,-8,0)$.
  \end{description}
\end{frame}

\begin{frame}{Backup: why LSTD(1) = Monte-Carlo regression}
  With $\lambda = \gamma = 1$, $\vz_t = \sum_{i\le t}\vphi_i$ and $\vphi_{T} = 0$ at termination:
  \[
    A = \sum_t \vz_t(\vphi_t - \vphi_{t+1})^\top
      = \sum_i \vphi_i \sum_{t\ge i}(\vphi_t - \vphi_{t+1})^\top
      = \sum_i \vphi_i\vphi_i^\top
  \]
  \[
    \mathbf{b} = \sum_t \vz_t r_t = \sum_i \vphi_i \sum_{t\ge i} r_t = \sum_i \vphi_i G_i
  \]
  So $\vth = A^{-1}\mathbf{b}$ is ordinary least squares of $G_i$ on $\vphi_i$ --- built incrementally without storing trajectories.
\end{frame}

\begin{frame}{Backup: three views of LSTD($\lambda$)}
  \speaker{\memberC}
  \begin{enumerate}
    \item \textbf{$\lambda=0$:} exactly Bradtke \& Barto's LSTD (1996), with a much simpler derivation.
    \item \textbf{$\lambda=1$:} least-squares regression of features onto Monte-Carlo returns,
      built \emph{incrementally} (telescoping: $A = \sum_t \vphi_t\vphi_t^\top$, $\mathbf{b} = \sum_t \vphi_t G_t$).
    \item \textbf{Model-based view:} with one feature per state, LSTD(0) accumulates
      \[ A = N - C_{\text{count}}, \quad \mathbf{b} = \mathbf{s}
        \;\Rightarrow\; \vth = (N - C_{\text{count}})^{-1}\mathbf{s}, \]
      which is exactly \hl{``count transitions, build the empirical Markov chain, solve its Bellman equations''}.
  \end{enumerate}
  \vspace{0.3em}
  \begin{block}{Corridor example, one pass over the 8 episodes}
    LSTD(0): $\hat V(A)=0.75$ (empirical TD fixed point).\quad
    LSTD(1): $\hat V(A) = 0$ (= MC).\quad
    TD(0) gives $\hat V(A)\approx0.75$ after 2000 replays with $\alpha=0.01$ in this example.\\
    \small The model-based and Monte-Carlo equivalences have explicit unit tests.
  \end{block}
\end{frame}
